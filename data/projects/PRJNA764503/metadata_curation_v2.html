Step 4: Creating a Uniform Metadata Template

This step converts the raw SRA table into the standard metadata format used in our meta-analysis. The uniform metadata should include the following columns: 1. bioproject: Project ID from SRA or other database 2. sample_id: SRA sample/run ID. Unique id for each sample! 3. subject_id: subject id for each person if available. 4. age: Age of the subject if available. e.g. 35, 42 5. age_info: If the age is not explicitly stated but approximate age range information is available, mark it in this column. 6. bmi: Body Mass Index if available. 7. sex: Sex of the subject. Mandatory to write as M/F 8. sequencing_region: Targeted 16S rRNA region. 9. platform: Sequencing platform. 10. country: Country of sample origin. 11. site: Sampling site. For oral and gut sites, specify the exact sites as detailed as possible. 12. site_group: only accepts one of following three values: ‘oral’, ‘gut’, or ‘other’. 13. public_time: Data release date. 14. pcr_primers: PCR primers used. 15. title: Title of the associated publication. 16. disease_study: Whether the study involves disease samples. Only three values accepted: ‘yes’, ‘no’ or NA 17. status: Disease status for each sample if available, else record as NA. 18. matched_samples: Whether the subject has matched samples. Only accept ‘yes’ or ‘no’. 19. is_time_series: Whether the study is a time series study. Only accept ‘yes’ or ‘no’.

# Function to check if a column contains "pending"
has_pending <- function(col) {
  if (is.character(col) || is.factor(col)) {
    any(col == "pending", na.rm = TRUE)
  } else {
    FALSE
  }
}

Step 6: Visualize the Metadata

This step makes a quick set of plots for checking the distribution of key metadata fields.

# Select columns for visualization, just run
plot_data <- metadata %>%
  select(-bioproject, -sample_id, -subject_id, -title)

create_simple_plots <- function(data) {
  plots <- list()
  continuous_vars <- c("age", "bmi")

  for (col in colnames(data)) {
    if (all(is.na(data[[col]]))) {
      p <- ggplot() +
        annotate("text", x = 0.5, y = 0.5, label = "this column is NA", size = 6) +
        labs(title = col) +
        theme_void()
    } else if (col %in% continuous_vars && is.numeric(data[[col]])) {
      p <- ggplot(data, aes(x = .data[[col]])) +
        geom_histogram(fill = "blue", alpha = 0.6, bins = 20, na.rm = TRUE) +
        labs(title = col, x = col, y = "Count") +
        theme_minimal()
    } else {
      count_data <- as.data.frame(table(data[[col]], useNA = "ifany"))
      p <- ggplot(count_data, aes(x = reorder(Var1, -Freq), y = Freq)) +
        geom_col(fill = "red", alpha = 0.6) +
        labs(title = col, x = NULL, y = "Count") +
        theme_minimal() +
        theme(axis.text.x = element_text(angle = 15, hjust = 1))
    }
    plots[[col]] <- p
  }

  return(plots)
}

# Generate and arrange plots
plots <- create_simple_plots(plot_data)
p <- plot_grid(plots$age,
               plots$age_info,
               plots$bmi,
               plots$sex,
               plots$site,
               plots$site_group,
               plots$sequencing_region,
               plots$platform,
               plots$country,
               plots$pcr_primers,
               plots$status,
               plots$matched_samples,
               ncol = 3, align = "hv", axis = "lb")
ggsave("Metadata_Summary.png", plot = p, width = 8, height = 6, dpi = 150)
