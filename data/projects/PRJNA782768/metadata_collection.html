#每列格式说明

1、BioProject：Project ID like “PRJNA339212”

2、Run：sample id with a name such as “ERRXXXXXX” or “SRRXXXXXX”

3、sample_name： 数据提交时的样品编号，不是数据库分配的编号

4、Age

5、BMI

6、Sex：fill with “F”, “M”

7、PCR_primers such as: 515F-806R,514F-805R,341F-785R,8F-926R,515F-926R, this usually can be found in method section of a paper

8、Region such as V2,V4,V3-V4,V1-V5,V6-V8,V3-V5

9、Platform “such as: 454,Illumina MiSeq,Illumina HiSeq 2000,Illumina HiSeq,Illumi MiSeq,Illumina,Illumina HiSeq 2500,Illumina NovaSeq 6000,Illumina MiniSeq,Illumi HiSeq 2500,ILLUMINANextSeq2000”

10、Country：China,USA,Brazil

11、Site ：“saliva,tongue,oral mucosa,dental plaque,oral swab,tooth,subgingival plaque,tongue coating,oral wash,buccal mucosa,lower jaw mucosa,lower labial mucosa,upper jaw mucosa,Spain,oral cavity,hard palate swab,keratinized gingiva swab,palatine tonsils swab,tongue dorsum swab,supragingival plaque,gingival sulcus,periodontal pocket,carious material,Swab,Attached_Keratinized_gingiva,hard palate,tongue dorsum”

12、public_time ：数据第一次发布的时间（ReleaseDate） 格式：2015-10-15

13、title： paper title

14、 Disease_study fill with “YES” if the paper related to disease study,otherwise fill with “NO”

15、Status： if ”Disease_study“ is YES, fill with the exact disease information such as “PD”, “IBD”, “Dental_Caries”

# 自动创建图形
create_simple_plots <- function(data) {
  plots <- list()
  
  for (col in colnames(data)) {
    if (is.numeric(data[[col]])) {
      # 数值变量：直方图
      p <- ggplot(data, aes(x = .data[[col]])) +
        geom_histogram(fill = "blue", alpha = 0.6, bins = 20) +
        labs(title = col) +
        theme_minimal()
    } else {
      # 分类变量：柱状图
      count_data <- as.data.frame(table(data[[col]]))
      p <- ggplot(count_data, aes(x = reorder(Var1, -Freq), y = Freq)) +
        geom_col(fill = "red", alpha = 0.6) +
        labs(title = col, x = NULL, y = "Count") +
        theme_minimal() +
        theme(axis.text.x = element_text(angle = 45, hjust = 1))
    }
    plots[[col]] <- p
  }
  
  return(plots)
}

# 生成图形并排列
plots <- create_simple_plots(plot_data)
grid.arrange(grobs = plots, ncol = 3, top = "Metadata Summary")

print(plots)
